  const readListings = () => {
    try {
      const stored = JSON.parse(localStorage.getItem(LISTINGS_KEY));
      if (Array.isArray(stored) && stored.length) {
        const existingIds = new Set(stored.map(listing => listing.id));
        const missingSeeds = seedListings.filter(listing => !existingIds.has(listing.id));
        if (missingSeeds.length) {
          const mergedListings = [...missingSeeds, ...stored];
          localStorage.setItem(LISTINGS_KEY, JSON.stringify(mergedListings));
          return mergedListings;
        }
        return stored;
      }
    } catch (e) {}
    localStorage.setItem(LISTINGS_KEY, JSON.stringify(seedListings));
    return seedListings;
  };
  const writeListings = (listings) => localStorage.setItem(LISTINGS_KEY, JSON.stringify(listings));

  const readBookings = () => {
    try {
      const stored = JSON.parse(localStorage.getItem(BOOKINGS_KEY));
      return Array.isArray(stored) ? stored : [];
    } catch (e) { return []; }
  };
  const writeBookings = (bookings) => localStorage.setItem(BOOKINGS_KEY, JSON.stringify(bookings));

  const listTabBtn = document.getElementById("listTabBtn");
  const browsePanel = document.getElementById("browsePanel");
  const listPanel = document.getElementById("listPanel");

  const showTab = (tab) => {
    const isBrowse = tab === "browse";
    browsePanel.classList.toggle("active", isBrowse);
    listPanel.classList.toggle("active", !isBrowse);
    browseTabBtn.classList.toggle("active", isBrowse);
    listTabBtn.classList.toggle("active", !isBrowse);
    if (!isBrowse) closeBookingPanel();
  };

  listTabBtn.addEventListener("click", () => showTab("list"));

  try { currentUser = JSON.parse(localStorage.getItem("playboy_current_user")); } catch (e) {}

    document.getElementById("renterName").value = currentUser.name;
    document.getElementById("listOwnerName").value = currentUser.name;
    if (currentUser.email) {

  const typeFilterNav = document.getElementById("typeFilterNav");
  const cityFilterInput = document.getElementById("cityFilterInput");
  let activeType = "all";

  const renderGrid = () => {
    const listings = readListings();
    const cityQuery = cityFilterInput.value.trim().toLowerCase();
    const filtered = listings.filter(l =>
      (activeType === "all" || l.type === activeType) &&
      (cityQuery === "" || l.city.toLowerCase().includes(cityQuery))
    );

    if (filtered.length === 0) {
      rentalGrid.innerHTML = '<p class="rental-empty">No consoles match your filters yet. Try another type or city.</p>';
      return;
    }

    rentalGrid.innerHTML = filtered.map(l => `
      <div class="rental-card reveal is-visible">
        <div class="rental-card-top">
          <span class="rental-badge">Available</span>
          <span class="rental-type-badge">${l.type}</span>
        </div>
        <h3>${l.consoleName}</h3>
        <p class="rental-meta">${l.condition} · ${l.city}</p>
        <p class="rental-owner">Listed by <strong>${l.ownerName}</strong></p>
        <div class="rental-price-row">

      typeFilterNav.querySelectorAll(".filter-btn").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      activeType = btn.dataset.type;
      renderGrid();
    });

  const bookingTitle = document.getElementById("bookingTitle");
  const bookingMeta = document.getElementById("bookingMeta");
  const bookingMinHours = document.getElementById("bookingMinHours");
  const bookingForm = document.getElementById("bookingForm");
  const bookingConfirmBox = document.getElementById("bookingConfirmBox");
  const hoursValueEl = document.getElementById("hoursValue");
  const controllersValueEl = document.getElementById("controllersValue");

  let activeListing = null;
  let hours = 1;
  let extraControllers = 0;

  const closeBookingPanel = () => {
    bookingPanel.classList.remove("active");
    bookingConfirmBox.classList.remove("active");
    bookingForm.style.display = "";
    activeListing = null;
  };
  document.getElementById("bookingCloseBtn").addEventListener("click", closeBookingPanel);

  const updateSummary = () => {
    if (!activeListing) return;
    const rentalCost = activeListing.hourlyRate * hours;
    const controllerCost = extraControllers * activeListing.extraControllerPrice;
    const total = rentalCost + controllerCost;
    hoursValueEl.textContent = hours;
    controllersValueEl.textContent = extraControllers;
    document.getElementById("summaryRateLabel").textContent = `${inr(activeListing.hourlyRate)}/hr × ${hours} hr${hours > 1 ? "s" : ""}`;
    document.getElementById("summaryRentalCost").textContent = inr(rentalCost);
    document.getElementById("summaryControllerLabel").textContent = `Extra controllers (${extraControllers})`;
    document.getElementById("summaryControllerCost").textContent = inr(controllerCost);
    document.getElementById("summaryDepositRow").textContent = `+ ${inr(activeListing.deposit)} refundable security deposit (paid separately)`;
    document.getElementById("summaryTotal").textContent = inr(total);
  };

  document.getElementById("hoursMinus").addEventListener("click", () => {
    if (!activeListing) return;
    if (hours > activeListing.minHours) { hours--; updateSummary(); }
  });
  document.getElementById("hoursPlus").addEventListener("click", () => { hours++; updateSummary(); });
  document.getElementById("controllersMinus").addEventListener("click", () => {
    if (extraControllers > 0) { extraControllers--; updateSummary(); }
  });
  document.getElementById("controllersPlus").addEventListener("click", () => { extraControllers++; updateSummary(); });

  rentalGrid.addEventListener("click", (e) => {
    const btn = e.target.closest(".rental-book-btn");
    if (!btn) return;
    const listings = readListings();
    activeListing = listings.find(l => l.id === btn.dataset.id);
    if (!activeListing) return;

    hours = activeListing.minHours;
    extraControllers = 0;
    bookingTitle.textContent = `Rent: ${activeListing.consoleName}`;
    bookingMeta.textContent = `${activeListing.condition} · ${activeListing.city} · Listed by ${activeListing.ownerName}`;
    bookingMinHours.textContent = activeListing.minHours;
    bookingForm.style.display = "";
    bookingConfirmBox.classList.remove("active");
    updateSummary();

    bookingConfirmBox.innerHTML = `
      <strong>Request sent!</strong> Your rental request for <strong>${activeListing.consoleName}</strong>
      on ${booking.renterDate || "the selected date"} for ${hours} hour${hours > 1 ? "s" : ""}
      (${extraControllers} extra controller${extraControllers === 1 ? "" : "s"}) has been recorded.<br><br>
      <strong>Total payable to owner:</strong> ${inr(total)} + ${inr(activeListing.deposit)} refundable deposit<br>
      <strong>Owner contact:</strong> ${activeListing.ownerName} · ${activeListing.ownerPhone}
    `;

    if (window.showMessage) window.showMessage("Rental request submitted", "success");
    renderMyActivity();
  });

  // ---- List a console ----
  const listForm = document.getElementById("listForm");
  listForm.addEventListener("submit", (e) => {
    e.preventDefault();

    const newListing = {
      id: "RENT-" + Math.floor(Math.random() * 1000000),
      consoleName: document.getElementById("listConsoleName").value.trim(),
      type: document.getElementById("listConsoleType").value,
      condition: document.getElementById("listCondition").value,

      ownerName: document.getElementById("listOwnerName").value.trim(),
      ownerPhone: document.getElementById("listOwnerPhone").value.trim(),
      notes: document.getElementById("listNotes").value.trim(),
      createdAt: new Date().toISOString()
    };

    const listings = readListings();
    listings.unshift(newListing);
    writeListings(listings);

    listForm.reset();
    document.getElementById("listHourlyRate").value = 250;
    document.getElementById("listMinHours").value = 2;
    document.getElementById("listIncludedControllers").value = 1;
    document.getElementById("listExtraControllerPrice").value = 100;
    document.getElementById("listDeposit").value = 1000;

    if (window.showMessage) window.showMessage("Your console is now listed", "success");
    renderGrid();
    renderMyActivity();
    showTab("browse");
  });

  // ---- My activity ----
  const renderMyActivity = () => {
    const listings = readListings().filter(l => !l.id.startsWith("RENT-SEED-"));

    document.getElementById("listingCount").textContent = listings.length;
    document.getElementById("bookingCount").textContent = bookings.length;
    document.getElementById("listingPanelCount").textContent = `${listings.length} active`;
    document.getElementById("bookingPanelCount").textContent = `${bookings.length} request${bookings.length === 1 ? "" : "s"}`;

    const myListingsEl = document.getElementById("myListings");
    myListingsEl.innerHTML = listings.length === 0
      ? '<div class="activity-empty"><div><span class="activity-empty-icon">⌁</span><strong>Your console collection starts here</strong><span>List a console to start earning from it.</span></div></div>'
      : listings.map(l => `
        <div class="rental-activity-card">
          <div class="activity-card-top">
            <strong class="activity-card-title">${l.consoleName}</strong>
            <span class="activity-card-type">${l.type}</span>
          </div>
          <div class="activity-card-meta">${l.condition} condition · ${l.city} · ${l.delivery === "Pickup Only" ? "Pickup" : l.delivery === "Can Deliver" ? "Delivery" : "Pickup & delivery"}</div>
          <div class="activity-card-footer">
            <span class="activity-card-value">${inr(l.hourlyRate)}<small>/hr</small></span>
            <span class="rental-status live">Live</span>
          </div>
        </div>
      `).join("");

    const myBookingsEl = document.getElementById("myBookings");
    myBookingsEl.innerHTML = bookings.length === 0
      ? '<div class="activity-empty"><div><span class="activity-empty-icon">◈</span><strong>No rental requests yet</strong><span>Choose a console above when you are ready to play.</span></div></div>'
      : bookings.map(b => `
        <div class="rental-activity-card">
          <div class="activity-card-top">
            <strong class="activity-card-title">${b.consoleName}</strong>
            <span class="activity-card-type">${b.renterDate || "Date pending"}</span>
          </div>
          <div class="activity-card-meta">${b.hours} hr${b.hours > 1 ? "s" : ""} rental · ${b.extraControllers} extra controller${b.extraControllers === 1 ? "" : "s"}</div>
          <div class="activity-card-footer">
            <span class="activity-card-value">${inr(b.total)} total</span>
            <span class="rental-status">${b.status}</span>
          </div>
        </div>
      `).join("");
  };

  renderGrid();
  renderMyActivity();
});
